        <script>
            // Form submission handling integrated with Google Sheets URL fetch routing
            document.addEventListener('DOMContentLoaded', function () {
                const form = document.getElementById('contact-form');
                const success = document.getElementById('form-success');
                const errorBox = document.getElementById('form-error');
                const submitBtn = document.getElementById('submit-btn');
                const submitBtnText = document.getElementById('submit-btn-text');

                if (form) {
                    form.addEventListener('submit', function (e) {
                        e.preventDefault();

                        // Set visual loading state
                        submitBtn.disabled = true;
                        if (submitBtnText) submitBtnText.innerText = 'Sending...';
                        success.classList.add('hidden');
                        if (errorBox) errorBox.classList.add('hidden');

                        const formData = new FormData(form);
                        const urlParams = new URLSearchParams();

                        formData.forEach((value, key) => {
                            urlParams.append(key, value);
                        });

                        // Uses GET query formatting to prevent CORS issues from local systems
                        fetch(form.action + '?' + urlParams.toString(), {
                            method: 'GET',
                            mode: 'no-cors'
                        })
                            .then(() => {
                                // Success callback triggers visual confirmation layout block
                                success.classList.remove('hidden');
                                form.reset();

                                setTimeout(() => {
                                    success.classList.add('hidden');
                                }, 5000);
                            })
                            .catch(error => {
                                console.error('Error:', error);
                                if (errorBox) errorBox.classList.remove('hidden');
                            })
                            .finally(() => {
                                // Reset submit button configurations
                                submitBtn.disabled = false;
                                if (submitBtnText) {
                                    submitBtnText.innerHTML = `
                            Send Message
                            <svg class="w-4 h-4 ml-2 group-hover:translate-x-1 transition-transform" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M14 5l7 7m0 0l-7 7m7-7H3"></path>
                            </svg>
                        `;
                                }
                            });
                    });
                }
            });
        </script>
